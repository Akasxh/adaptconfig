% ── 6. THE MOAT ──────────────────────────────────────────────────────────────
\begin{frame}{Why we win in \emph{banking}, not just \emph{integration}}
\vspace{0.2em}
\begin{columns}[T]
\begin{column}{0.48\textwidth}
\textbf{1. Rules override the LLM.}\\
{\small Deterministic synonym + fuzzy + Jaccard match runs after every LLM call. The LLM is allowed to suggest --- never to silently decide. \textcolor{teal}{No hallucination reaches production.}}

\vspace{0.6em}
\textbf{2. Confidence-driven validation.}\\
{\small Every field carries a score. Below threshold $\to$ human review. \textcolor{teal}{Banking compliance demands traceable provenance --- we ship it by default.}}
\end{column}
\begin{column}{0.48\textwidth}
\textbf{3. Indian fintech depth, baked in.}\\
{\small 100+ domain synonyms (CIBIL score $=$ credit\_score, Aadhaar $=$ uidai\_id, $\ldots$). Pre-built adapters for the 8 services every NBFC actually uses.}

\vspace{0.6em}
\textbf{4. Banking-grade audit \& replay.}\\
{\small Hash-chain tamper detection on every external call $\cdot$ 8-step pre-deploy simulator $\cdot$ versioned configs with one-click rollback.}
\end{column}
\end{columns}
\end{frame}
